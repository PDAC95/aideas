```sql
-- V1: 8 functional areas
SELECT count(*) AS active_areas FROM public.functional_areas WHERE is_active = true;
-- expect 8

SELECT slug, label_en, label_es, sort_order
FROM public.functional_areas WHERE is_active = true
ORDER BY sort_order;
-- expect: ventas, marketing, atencion-al-cliente, documentos, productividad,
--         reportes, agentes-ia, integraciones-seguridad

-- V2: 50 active scenarios
SELECT count(*) AS active_scenarios FROM public.scenarios WHERE is_active = true;
-- expect 50

-- V3: ~135 mappings, no orphans
SELECT count(*) AS total_mappings FROM public.scenario_templates;
-- expect 100-160

SELECT count(*) AS orphan_scenarios
FROM public.scenarios s WHERE s.is_active = true
  AND NOT EXISTS (SELECT 1 FROM public.scenario_templates st WHERE st.scenario_id = s.id);
-- expect 0

-- V4: Per-area distribution
SELECT fa.slug, count(s.id) AS scenarios
FROM public.functional_areas fa
LEFT JOIN public.scenarios s ON s.functional_area_id = fa.id AND s.is_active = true
WHERE fa.is_active = true
GROUP BY fa.slug, fa.sort_order
ORDER BY fa.sort_order;
-- expect 8 rows, floor 3 per area, sum = 50

-- V5: Bilingual completeness
SELECT
  count(*) FILTER (WHERE pain_headline_en IS NULL OR pain_headline_en = '') AS missing_en,
  count(*) FILTER (WHERE pain_headline_es IS NULL OR pain_headline_es = '') AS missing_es,
  count(*) FILTER (WHERE pain_body_en IS NULL OR pain_body_en = '') AS missing_body_en,
  count(*) FILTER (WHERE pain_body_es IS NULL OR pain_body_es = '') AS missing_body_es,
  count(*) FILTER (WHERE impact_label_en IS NULL OR impact_label_en = '') AS missing_impact_en,
  count(*) FILTER (WHERE impact_label_es IS NULL OR impact_label_es = '') AS missing_impact_es
FROM public.scenarios WHERE is_active = true;
-- expect all zeros

-- V6: Hours estimates are integers 1-20
SELECT count(*) AS bad_hours
FROM public.scenarios
WHERE is_active = true
  AND (typical_hours_per_week IS NULL
       OR typical_hours_per_week < 1
       OR typical_hours_per_week > 20
       OR typical_hours_per_week != floor(typical_hours_per_week));
-- expect 0

-- V7: Anon RLS chain end-to-end
SET ROLE anon;
SELECT
  count(DISTINCT fa.id) AS visible_areas,
  count(DISTINCT s.id) AS visible_scenarios,
  count(st.template_id) AS visible_mappings
FROM public.functional_areas fa
LEFT JOIN public.scenarios s ON s.functional_area_id = fa.id
LEFT JOIN public.scenario_templates st ON st.scenario_id = s.id;
RESET ROLE;
-- expect: visible_areas=8, visible_scenarios=50, visible_mappings = total_mappings from V3
```
