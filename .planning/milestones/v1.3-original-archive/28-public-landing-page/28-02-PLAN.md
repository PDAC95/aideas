1. **Query helper** (`web/src/lib/landing/queries.ts`):
   ```ts
   import { createClient } from "@/lib/supabase/server";

   export type PublicFunctionalArea = {
     id: string;
     slug: string;
     label_en: string;
     label_es: string;
     description_en: string | null;
     description_es: string | null;
     sort_order: number;
   };

   export async function fetchPublicFunctionalAreas(): Promise<PublicFunctionalArea[]> {
     const supabase = await createClient();
     const { data, error } = await supabase
       .from("functional_areas")
       .select("id, slug, label_en, label_es, description_en, description_es, sort_order")
       .eq("is_active", true)
       .order("sort_order", { ascending: true });
     if (error) {
       console.error("[landing/queries] fetchPublicFunctionalAreas failed", error);
       return [];
     }
     return data ?? [];
   }
   ```
   This uses the public anon key via the existing `createClient()` server helper. The anon SELECT policy (Phase 26) returns the 8 active areas without authentication.

2. **ServicesSection** (`web/src/components/landing/sections/services-section.tsx`) — Server Component:
   ```tsx
   import Link from "next/link";
   import { getLocale, getTranslations } from "next-intl/server";
   import { Card } from "@/components/ui/card";
   import { Button } from "@/components/ui/button";
   import { fetchPublicFunctionalAreas } from "@/lib/landing/queries";
   import { ShoppingCart, Megaphone, Headset, FileText, Zap, BarChart3, Bot, Shield } from "lucide-react";

   const ICONS: Record<string, React.ComponentType<{ className?: string }>> = {
     "ventas": ShoppingCart,
     "marketing": Megaphone,
     "atencion-cliente": Headset,
     "documentos": FileText,
     "productividad": Zap,
     "reportes": BarChart3,
     "agentes-ia": Bot,
     "integraciones-seguridad": Shield,
   };

   export async function ServicesSection() {
     const t = await getTranslations("landing.services");
     const locale = await getLocale();
     const areas = await fetchPublicFunctionalAreas();

     return (
       <section className="py-20 container mx-auto px-4">
         <header className="max-w-2xl mx-auto text-center mb-12">
           <h2 className="text-3xl lg:text-4xl font-bold">{t("title")}</h2>
           <p className="mt-4 text-muted-foreground">{t("subtitle")}</p>
         </header>

         <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
           {areas.map((area) => {
             const Icon = ICONS[area.slug] ?? Zap;
             const label = locale === "es" ? area.label_es : area.label_en;
             const description = locale === "es" ? area.description_es : area.description_en;
             return (
               <Link key={area.id} href={`/catalog/${area.slug}`} className="block group focus:outline-none focus-visible:ring-2 focus-visible:ring-ring rounded-md">
                 <Card className="h-full p-6 transition-colors group-hover:border-primary">
                   <Icon className="h-8 w-8 text-primary mb-4" aria-hidden="true" />
                   <h3 className="font-semibold text-lg mb-2">{label}</h3>
                   {description && <p className="text-sm text-muted-foreground">{description}</p>}
                 </Card>
               </Link>
             );
           })}
         </div>

         <div className="mt-10 text-center">
           <Link href="/catalog">
             <Button variant="outline" size="lg">{t("ctaAll")}</Button>
           </Link>
         </div>
       </section>
     );
   }
   ```
   Notes:
   - Entire Card wrapped in `<Link>` so the whole card is clickable (CONTEXT.md decision)
   - Grid: 1 col <640px, 2 cols 640-1023px, 4 cols ≥1024px → produces the 4×2 layout from CONTEXT.md
   - Icon selection: keyed by slug, falls back to `Zap` to avoid runtime crash if a new slug ships
   - `label` / `description` picks locale-appropriate column from the row (this is content data, not UI chrome, so it's NOT in next-intl — it comes from the Supabase seed, which already has both languages)
   - `t("ctaAll")` is "Ver catálogo completo" / "See the full catalog"
